5606 p->tr2 = pow2(p->tr2);
5607 case PLUS:
5608 case MINUS:
5609 case AND:
5610 case NAND:
5611 case OR:
5612 case EXOR:
5613 case LSHIFT:
5614 case RSHIFT:
5615 p1 = p->tr2->tr2;
5616 if (xdcalc(p1) > 12
5617 || p1->op==NAME
5618 &&(p1->nloc==p->tr1->nloc
5619 || p1->regno==p->tr1->nloc))
5620 return(0);
5621 p1 = p->tr2;
5622 p->tr2 = p1->tr1;
5623 if (p1->tr1->op!=NAME
5624 || p1->tr1->class!=REG
5625 || p1->tr1->nloc!=p->tr1->nloc)
5626 rcexpr(p, efftab, reg);
5627 p->tr2 = p1->tr2;
5628 p->op = p1->op + ASPLUS - PLUS;
5629 *treep = p;
5630 return(1);
5631 }
5632 goto OK;
5633
5634 case ASTIMES:
5635 case ASDIV:
5636 if (!ispow2(p))
5637 return(0);
5638 case ASPLUS:
5639 case ASMINUS:
5640 case ASSAND:
5641 case ASSNAND:
5642 case ASOR:
5643 case ASXOR:
5644 case DECBEF:
5645 case INCBEF:
5646 OK:
5647 if (table==cctab||table==cregtab)
5648 reg =+ 020;
5649 rcexpr(optim(p), efftab, ~reg);
5650 *treep = p1;
5651 return(1);
5652 }
5653 return(0);
5654}

5656/*
5657 * Delay handles postfix ++ and --
5658 * It observes that "x + y++" is better
5659 * treated as "x + y; y++".
5660 * If the operator is ++ or -- itself,
5661 * it calls rcexpr to load the operand, letting
5662 * the calling instance of rcexpr to do the
5663 * ++ using efftab.
5664 * Otherwise it uses sdelay to search for inc/dec
5665 * among the operands.
5666 */
5667delay(treep, table, reg)
5668struct tnode **treep;
5669{
5670 register struct tnode *p, *p1;
5671 register r;
5672
5673 p = *treep;
5674 if (table!=efftab && (p->op==INCAFT||p->op==DECAFT)
5675 && p->tr1->op==NAME) {
5676 return(1+rcexpr(p->tr1, table, reg));
5677 }
5678 p1 = 0;
5679 if (opdope[p->op]&BINARY)
5680 p1 = sdelay(&p->tr2);
5681 if (p1==0)
5682 p1 = sdelay(&p->tr1);
5683 if (p1) {
5684 r = rcexpr(optim(p), table, reg);
5685 *treep = p1;
5686 return(r+1);
5687 }
5688 return(0);
5689}
5690
5691sdelay(ap)
5692struct tnode **ap;
5693{
5694 register struct tnode *p, *p1;
5695
5696 p = *ap;
5697 if ((p->op==INCAFT||p->op==DECAFT) && p->tr1->op==NAME) {
5698 *ap = ncopy(p->tr1);
5699 return(p);
5700 }
5701 if (p->op==STAR || p->op==PLUS)
5702 if (p1=sdelay(&p->tr1))
5703 return(p1);
5704 if (p->op==PLUS)
5705 return(sdelay(&p->tr2));
5706 return(0);
5707}

5732chkleaf(atree, table, reg)
