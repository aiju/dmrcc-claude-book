7020distrib(list)

7022{
7023/*
7024 * Find a list member of the form c1c2*x such
7025 * that c1c2 divides no other such constant, is divided by
7026 * at least one other (say in the form c1*y), and which has
7027 * fewest divisors. Reduce this pair to c1*(y+c2*x)
7028 * and iterate until no reductions occur.
7029 */

7033 struct tnode **dividend, **divisor;
7034 struct tnode **maxnod, **mindiv;
7035
7036 loop:
7037 maxnod = &list->llist[list->nextl];
7038 ndmaj = 1000;
7039 dividend = 0;
7040 for (p1 = list->llist; p1 <= maxnod; p1++) {
7041 if ((*p1)->op!=TIMES || (*p1)->tr2->op!=CON)
7042 continue;
7043 ndmin = 0;
7044 for (p2 = list->llist; p2 <= maxnod; p2++) {
7045 if (p1==p2 || (*p2)->op!=TIMES || (*p2)->tr2->op!=CON)
7046 continue;
7047 if ((*p1)->tr2->value == (*p2)->tr2->value) {
7048 (*p2)->tr2 = (*p1)->tr1;
7049 (*p2)->op = PLUS;
7050 (*p1)->tr1 = (*p2);
7051 *p1 = optim(*p1);
7052 squash(p2, maxnod);
7053 list->nextl--;
7054 goto loop;
7055 }
7056 if (((*p2)->tr2->value % (*p1)->tr2->value) == 0)
7057 goto contmaj;
7058 if (((*p1)->tr2->value % (*p2)->tr2->value) == 0) {
7059 ndmin++;
7060 mindiv = p2;
7061 }
7062 }
7063 if (ndmin > 0 && ndmin < ndmaj) {
7064 ndmaj = ndmin;
7065 dividend = p1;
7066 divisor = mindiv;
7067 }
7068 contmaj:;
7069 }
7070 if (dividend==0)

7072 t = list->nlist[--list->nextn];
7073 p1 = dividend;
7074 p2 = divisor;
7075 t->op = PLUS;
7076 t->type = (*p1)->type;
7077 t->tr1 = (*p1);

7088 }
