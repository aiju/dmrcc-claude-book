1810/*
1811 * Read an expression and return a pointer to its tree.
1812 * It's the classical bottom-up, priority-driven scheme.
1813 * The initflg prevents the parse from going past
1814 * "," or ":" because those delimitesrs are special
1815 * in initializer (and some other) expressions.
1816 */
1817tree()
1818{
1819#define SEOF 200
1820#define SSIZE 20
1821 int *op, opst[SSIZE], *pp, prst[SSIZE];
1822 register int andflg, o;
1823 register struct hshtab *cs;
1824 int p, ps, os;
1825
1826 space = treespace;
1827 op = opst;
1828 pp = prst;
1829 cp = cmst;
1830 *op = SEOF;
1831 *pp = 06;
1832 andflg = 0;
1833
1834advanc:
1835 switch (o=symbol()) {
1836
1837 case NAME:
1838 cs = csym;
1839 if (cs->hclass==0 && cs->htype==0)
1840 if(nextchar()=='(') {
1841 /* set function */
1842 cs->hclass = EXTERN;
1843 cs->htype = FUNC;
1844 } else if (initflg)
1845 cs->hclass = EXTERN;
1846 else {
1847 /* set label */
1848 cs->htype = ARRAY;
1849 if (cs->hoffset==0)
1850 cs->hoffset = isn++;
1851 }
1852 *cp++ = copname(cs);
1853 goto tand;
1854
1855 case FCON:
1856 if (!initflg)
1857 outcode("BBNB1N1N1N1N0B", DATA,LABEL,
1858 cval, WDATA, fcval, PROG);
1859
1860 case CON:
1861 case SFCON:
1862 *cp++ = block(1,o,(o==CON?INT:DOUBLE),0,cval);
1863 goto tand;
1864
1865 /* fake a static char array */
1866 case STRING:
1867 putstr(cval);
1868 *cp++ = block(3, NAME, ARRAY+CHAR,0,STATIC,0,cval);
1869
1870tand:
1871 if(cp>=cmst+cmsiz) {
1872 error("Expression overflow");
1873 exit(1);
1874 }
1875 if (andflg)
1876 goto syntax;
1877 andflg = 1;
1878 goto advanc;
1879
1880 case INCBEF:
1881 case DECBEF:
1882 if (andflg)
1883 o =+ 2;
1884 goto oponst;
1885
1886 case COMPL:
1887 case EXCLA:
1888 case SIZEOF:
1889 if (andflg)
1890 goto syntax;
1891 goto oponst;
1892
1893 case MINUS:
1894 if (!andflg) {
1895 if ((peeksym=symbol())==FCON) {
1896 fcval = - fcval;
1897 goto advanc;
1898 }
1899 if (peeksym==SFCON) {
1900 fcval = - fcval;
1901 cval =^ 0100000;
1902 goto advanc;
1903 }
1904 o = NEG;
1905 }
1906 andflg = 0;
1907 goto oponst;
1908
1909 case AND:
1910 case TIMES:
1911 if (andflg)
1912 andflg = 0; else
1913 if(o==AND)
1914 o = AMPER;
1915 else
1916 o = STAR;
1917 goto oponst;
1918
1919 case LPARN:
1920 if (andflg) {
1921 o = symbol();
1922 if (o==RPARN)
1923 o = MCALL;
1924 else {
1925 peeksym = o;
1926 o = CALL;
1927 andflg = 0;
1928 }
1929 }

1931
1932 case RBRACK:
1933 case RPARN:
1934 if (!andflg)
1935 goto syntax;
1936 goto oponst;
1937
1938 case DOT:
1939 case ARROW:
1940 mosflg++;
1941 break;
1942
1943 }
1944 /* binaries */
1945 if (!andflg)
1946 goto syntax;
1947 andflg = 0;
1948
1949oponst:
1950 p = (opdope[o]>>9) & 077;
1951 if ((o==COMMA || o==COLON) && initflg)
1952 p = 05;
1953opon1:
1954 ps = *pp;
1955 if (p>ps || p==ps && (opdope[o]&RASSOC)!=0) {

1957
1958 case INCAFT:
1959 case DECAFT:
1960 p = 37;
1961 break;
1962 case LPARN:
1963 case LBRACK:
1964 case CALL:
1965 p = 04;
1966 }
1967 if (op >= &opst[SSIZE-1]) {
1968 error("expression overflow");
1969 exit(1);
1970 }

1972 *++pp = p;

1974 }

1977

1980 build(0); /* flush conversions */
1981 return(*--cp);
1982
1983 case CALL:

1988
1989 case MCALL:

1991 os = CALL;
1992 break;
1993
1994 case INCBEF:
1995 case INCAFT:
1996 case DECBEF:
1997 case DECAFT:
1998 *cp++ = block(1, CON, INT, 0, 1);
1999 break;
2000
2001 case LPARN:
2002 if (o!=RPARN)
2003 goto syntax;
2004 goto advanc;
2005
2006 case LBRACK:

2008 goto syntax;
